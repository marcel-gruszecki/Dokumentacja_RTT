1//! # Zarządzanie Sesją Sprzętową JTAG/RTT
2//!
3//! Projekt: RTT-Task Analyser
4//! Autor: Marcel Gruszecki (UAM)
5//! Moduł: `communication::session`
6//! Opis: Odpowiada za inicjalizację połączenia JTAG za pośrednictwem biblioteki `probe-rs`.
7//!       Skanuje pamięć HP SRAM mikrokontrolera ESP32-P4 w celu lokalizacji bloku kontrolnego RTT,
8//!       a następnie w sposób nieblokujący pobiera strumień bajtów z dedykowanego kanału telemetrii.
9
10use std::error::Error;
11use std::time::Duration;
12use std::thread;
13
14use probe_rs::{Permissions, config::TargetSelector, probe::list::Lister};
15use probe_rs::rtt::{Rtt, ScanRegion};
16
17use super::protocol::{FrameReader, PKT_LEN, EVT_IN, EVT_OUT, task_name};
18use super::scheduler::{TaskEvent, EventKind};
19
20// Adresy pamięci High-Performance SRAM dla ESP32-P4 (768 KB)
21const RTT_SCAN_START: u64 = 0x4ff0_0000;
22const RTT_SCAN_END:   u64 = 0x4ffc_0000;
23const RTT_CHANNEL:    usize = 1;
24
25/// Struktura reprezentująca aktywną sesję debugowania sprzętowego.
26pub struct TracerSession {
27    session: probe_rs::Session,
28    rtt:     Rtt,
29    framer:  FrameReader,
30}
31
32impl Drop for TracerSession {
33    /// Bezpiecznie zamyka sesję, czyści sprzętowe punkty przerwań i wznawia pracę procesora.
34    fn drop(&mut self) {
35        println!("\n[JTAG] Zamykanie sesji, czyszczenie rejestrów i wznawianie rdzeni...");
36        if let Ok(mut core) = self.session.core(0) {
37            let _ = core.clear_all_hw_breakpoints();
38            let _ = core.run();
39        }
40    }
41}
42
43impl TracerSession {
44    /// Łączy się z programatorem JTAG, inicjalizuje procesor i lokalizuje blok pamięci RTT.
45    pub fn connect(chip: &str) -> Result<Self, Box<dyn Error>> {
46        let lister = Lister::new();
47        let probes = lister.list_all();
48        if probes.is_empty() {
49            return Err("nie znaleziono programatora JTAG (sprawdź podłączenie USB)".into());
50        }
51
52        println!("Znalezione programatory:");
53        for (i, info) in probes.iter().enumerate() {
54            println!("  [{}] {:?}", i, info);
55        }
56
57        let probe = probes[0].open()?;
58        let mut session = probe.attach(
59            TargetSelector::Unspecified(chip.to_string()),
60            Permissions::default(),
61        )?;
62
63        // Uruchomienie rdzenia i oczekiwanie na start oprogramowania układowego
64        {
65            let mut core = session.core(0)?;
66            core.run()?;
67        }
68        println!("Rdzeń uruchomiony, czekam 500ms na init firmware...");
69        thread::sleep(Duration::from_millis(500));
70
71        // Skanowanie wskazanego regionu HP SRAM w poszukiwaniu bloku RTT
72        let scan = ScanRegion::range(RTT_SCAN_START..RTT_SCAN_END);
73        println!("Szukam bloku RTT w HP SRAM ({:#010x}..{:#010x})...", RTT_SCAN_START, RTT_SCAN_END);
74
75        let rtt = loop {
76            let mut core = session.core(0)?;
77            match Rtt::attach_region(&mut core, &scan) {
78                Ok(r) => {
79                    println!("Znaleziono blok RTT pod adresem: {:#010x}", r.ptr());
80                    core.run()?;
81                    break r;
82                }
83                Err(e) => {
84                    eprintln!("\r  [{:?}] próbuję ponownie...", e);
85                    thread::sleep(Duration::from_millis(200));
86                }
87            }
88        };
89
90        Ok(Self { session, rtt, framer: FrameReader::new() })
91    }
92
93    /// Pobiera dostępne bajty z bufora kołowego RTT i dekoduje je na wektor zdarzeń.
94    pub fn read_events(&mut self) -> Result<Vec<TaskEvent>, Box<dyn Error>> {
95        // Zwiększony bufor do 512 pakietów zapobiega gubieniu danych przy szybkich przełączeniach kontekstu
96        let mut raw_buf = [0u8; PKT_LEN * 512];
97
98        let session = &mut self.session;
99        let rtt     = &mut self.rtt;
100
101        let mut core = session.core(0)?;
102        if core.core_halted()? {
103            eprintln!("[warn] rdzeń był zatrzymany — wznawiam");
104            core.run()?;
105        }
106
107        // Odczyt surowego strumienia bajtów z kanału telemetrycznego przez interfejs JTAG
108        let n = rtt
109            .up_channels()
110            .get_mut(RTT_CHANNEL)
111            .ok_or("brak kanału RTT 1 — czy tracer_init() zostało wywołane?")?
112            .read(&mut core, &mut raw_buf)?;
113
114        drop(core);
115
116        let mut events = Vec::new();
117        if n > 0 {
118            self.framer.push(&raw_buf[..n]);
119            // Pętla dekodująca kompletne pakiety binarne
120            while let Some(pkt) = self.framer.next_packet() {
121                let kind = match pkt.event_type {
122                    EVT_IN  => EventKind::SwitchedIn,
123                    EVT_OUT => EventKind::SwitchedOut,
124                    other   => {
125                        eprintln!("[warn] Uszkodzona ramka RTT lub nieznany typ: {other:#x}");
126                        continue;
127                    }
128                };
129                events.push(TaskEvent {
130                    task_id:      pkt.task_id,
131                    name:         task_name(&pkt.name_raw),
132                    core_id:      pkt.core_id,
133                    priority:     pkt.priority,
134                    timestamp_us: pkt.timestamp_us,
135                    kind,
136                });
137            }
138        }
139        Ok(events)
140    }
141}
