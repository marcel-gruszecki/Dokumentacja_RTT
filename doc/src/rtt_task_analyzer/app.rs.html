19/// Główna struktura przechowująca stan aplikacji graficznej.
20pub struct MyEguiApp {
21    current_panel: ui::top_panel::PANEL,
22    analysis_struct: ui::analysis_panel::AnalysisPanel,
23    tracer: TracerHandle,
24    chip: String,
25    scheduling_results: Vec<SchedulingResult>,
26}
27
28impl Default for MyEguiApp {
29    fn default() -> Self {
30        MyEguiApp {
31            current_panel: Default::default(),
32            analysis_struct: Default::default(),
33            tracer: TracerHandle::new(),
34            chip: "esp32p4".to_string(),
35            scheduling_results: Vec::new(),
36        }
37    }
38}
39
40impl MyEguiApp {
41    pub fn new(_cc: &eframe::CreationContext<'_>) -> Self {
42        Self::default()
43    }
44}
45
46impl eframe::App for MyEguiApp {
47    fn update(&mut self, ctx: &egui::Context, _frame: &mut eframe::Frame) {
48        let status = self.tracer.status();
49
50        if matches!(status, ConnectionStatus::Connecting | ConnectionStatus::Connected) {
51            ctx.request_repaint_after(Duration::from_millis(100));
52        }
53
54        // Pobranie migawki danych z bazy wątku RTT
55        let (completed, schedule_data): (Vec<TaskExecution>, Vec<TaskScheduleData>) = {
56            let db = self.tracer.db.lock().unwrap();
57            (db.completed.clone(), db.schedule_data())
58        };
59
60        // Panel górny i akcje połączenia
61        let action = ui::top_panel::show_top_panel(ctx, &mut self.current_panel, &mut self.chip, &status);
62        match action {
63            ui::top_panel::TopPanelAction::Connect => {
64                self.tracer.connect(&self.chip.clone());
65            }
66            ui::top_panel::TopPanelAction::Disconnect => {
67                self.tracer.disconnect();
68                self.scheduling_results.clear();
69            }
70            ui::top_panel::TopPanelAction::None => {}
71        }
72
73        // Przełączanie widoków głównych
74        match self.current_panel {
75            ui::top_panel::PANEL::TASK => {
76                ui::task_panel::show_task_panel(ctx, &completed, &schedule_data);
77            }
78            ui::top_panel::PANEL::ANALYSIS => {
79                let action = ui::analysis_panel::show_analysis_panel(ctx, &mut self.analysis_struct, &self.scheduling_results);
80                match action {
81                    ui::analysis_panel::AnalysisAction::RunAll => {
82                        if !completed.is_empty() {
83                            self.scheduling_results = run_all(&completed);
84                        }
85                    }
86                    ui::analysis_panel::AnalysisAction::RunSelected(idx) => {
87                        if !completed.is_empty() {
88                            self.scheduling_results = vec![run_one(&completed, idx)];
89                        }
90                    }
91                    ui::analysis_panel::AnalysisAction::None => {}
92                }
93            }
94        }
95    }
96
97    fn on_exit(&mut self, _gl: Option<&eframe::glow::Context>) {
98        println!("[egui] Zamknięcie aplikacji, czyszczenie połączeń.");
99        self.tracer.disconnect();
100    }
101}
